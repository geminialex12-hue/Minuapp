    // Exponential Backoff Fetch Helper
    async function fetchWithRetry(url, options, retries = 3) {
      let delay = 1000;
      for (let i = 0; i <= retries; i++) {
        const res = await fetch(url, options);
        if (res.status === 429 && i < retries) {
          await new Promise(r => setTimeout(r, delay));
          delay *= 2;
          continue;
        }
        return res;
      }
    }

    async function callGeminiStandard(historySlice) {
      const apiKey = "";
      const url = `https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash:generateContent?key=${apiKey}`;
      const contents = historySlice.map(m => ({
        role: m.role === 'assistant' ? 'model' : 'user',
        parts: [{ text: m.content }]
      }));

      const res = await fetchWithRetry(url, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
          contents,
          systemInstruction: {
            parts: [{ text: 'Ты лаконичный, технически точный AI-ассистент в системе STAR // NOIR OS. Отвечай на русском языке.' }]
          }
        })
      });
      const data = await res.json();
      if (!res.ok) throw new Error(data?.error?.message || `Gemini HTTP ${res.status}`);
      const text = data?.candidates?.[0]?.content?.parts?.[0]?.text;
      if (!text) throw new Error('Пустой ответ Gemini');
      return text;
    }

    async function callGeminiDeepThink(historySlice) {
      const apiKey = "";
      const url = `https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash:generateContent?key=${apiKey}`;
      const contents = historySlice.map(m => ({
        role: m.role === 'assistant' ? 'model' : 'user',
        parts: [{ text: m.content }]
      }));

      const payload = {
        contents,
        systemInstruction: {
          parts: [{
            text: 'Ты аналитический движок глубоких раздумий STAR // NOIR OS. Сначала проведи пошаговый логический разбор задачи, проверь краевые случаи и гипотезы в поле "thinking", а затем сформулируй чёткий, исчерпывающий итоговый ответ в поле "answer". Отвечай строго на русском языке.'
          }]
        },
        generationConfig: {
          responseMimeType: 'application/json',
          responseSchema: {
            type: 'OBJECT',
            properties: {
              thinking: {
                type: 'STRING',
                description: 'Пошаговый ход мыслей, проверка фактов и анализ задачи по пунктам 1, 2, 3...'

              answer: {
                type: 'STRING',

              }
            },
            required: ['thinking', 'answer'],
            propertyOrdering: ['thinking', 'answer']
          }
        }
      };

      const res = await fetchWithRetry(url, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify(payload)
      });
      const data = await res.json();
      if (!res.ok) throw new Error(data?.error?.message || `Deep Think HTTP ${res.status}`);
      const rawJson = data?.candidates?.[0]?.content?.parts?.[0]?.text;
      if (!rawJson) throw new Error('Пустой ответ Deep Think');
      const parsed = JSON.parse(rawJson);
      return {
        thinking: parsed.thinking || 'Анализ завершён.',
        answer: parsed.answer || ''
      };
    }

          responseModalities: ['IMAGE'],
          imageConfig: { aspectRatio: '1:1' }
        }
      };

      const res = await fetchWithRetry(url, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify(payload)
      });
      const data = await res.json();
      if (!res.ok) throw new Error(data?.error?.message || `Image Studio HTTP ${res.status}`);
      const part = data?.candidates?.[0]?.content?.parts?.find(p => p.inlineData);
      if (!part || !part.inlineData?.data) {
        throw new Error('Модель не вернула изображение по данному запросу');
      }
      return `data:${part.inlineData.mimeType || 'image/png'};base64,${part.inlineData.data}`;
    }

    async function callOpenAICompatible(providerCfg, historySlice, extraHeaders = {}) {
      const messages = [
        { role: 'system', content: 'Ты лаконичный и точный AI-ассистент. Отвечай на русском языке.' },
        ...historySlice.map(m => ({ role: m.role, content: m.content }))
      ];

      const res = await fetch(providerCfg.endpoint, {
        method: 'POST',
        headers: {
          'Authorization': `Bearer ${providerCfg.key}`,
          'Content-Type': 'application/json',
          ...extraHeaders
        },
        body: JSON.stringify({
          model: providerCfg.model,
          messages
        })
      });

      const data = await res.json();
      if (!res.ok) throw new Error(data?.error?.message || `HTTP ${res.status}`);
      const text = data?.choices?.[0]?.message?.content;
      if (!text) throw new Error('Пустой ответ модели');
      return text;
    }

          document.getElementById('diagWebRtcHint').textContent = 'Утечка невозможна';
        } else {
          const pc = new RTCPeerConnection({ iceServers: [] });
          pc.createDataChannel('check');
          const offer = await pc.createOffer();
          await pc.setLocalDescription(offer);

          let status = 'mDNS Защита (.local)';
          let hint = 'Локальный IP замаскирован';

          await new Promise(resolve => {
            const t = setTimeout(() => { pc.close(); resolve(); }, 800);
            pc.onicecandidate = (ev) => {
              if (!ev || !ev.candidate) {
                clearTimeout(t);
                pc.close();
                resolve();
                return;
              }
              const cStr = ev.candidate.candidate || '';
              if (cStr.includes('.local')) {
                status = 'mDNS UUID (.local)';
                hint = 'Браузер скрывает локальный IP';
              } else {

          document.getElementById('diagWebRtcHint').textContent = hint;
